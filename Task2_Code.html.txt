    <!-- JavaScript for Billing Toggle -->
    <script>

        function changeBilling(type, button) {

            // Update toggle button
            document.querySelectorAll(".toggle-pill button")
                .forEach(btn => {
                    btn.classList.remove("active");
                });

            button.classList.add("active");

            // Update prices
            document.querySelectorAll(".plan-price")
                .forEach(price => {

                    if (type === "monthly") {

                        let value = price.dataset.monthly;

                        if (value === "Free") {
                            price.textContent = "Free";
                        } else {
                            price.innerHTML =
                                value + " <span>/mo</span>";
                        }

                    } else {

                        let value = price.dataset.annual;

                        if (value === "Free") {
                            price.textContent = "Free";
                        } else {
                            price.innerHTML =
                                value + " <span>/year</span>";
                        }

                    }

                });

        }

    </script>
